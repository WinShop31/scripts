[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
$ProgressPreference = 'SilentlyContinue'

$desktop = [Environment]::GetFolderPath("Desktop")
$targetFolder = Join-Path $desktop "Court_Modpack_1.16.5"

if (-not (Test-Path $targetFolder)) {
    New-Item -ItemType Directory -Path $targetFolder | Out-Null
    Write-Host "📁 Создана папка: $targetFolder" -ForegroundColor Yellow
}

# Список названий модов для автоматического поиска через API
$modList = @(
    "worldedit",
    "sodium",
    "lithium",
    "ferritecore",
    "lazydfu",
    "iris",
    "lambdynamiclights",
    "presence-footsteps",
    "sound-physics-fabric",
    "effective",
    "decorative-blocks",
    "supplementaries"
)

Write-Host "🚀 Поиск и скачивание модов под Fabric 1.16.5...`n" -ForegroundColor Cyan

foreach ($modSlug in $modList) {
    Write-Host "🔍 Поиск: $modSlug..." -NoNewline
    
    try {
        # Запрос к Modrinth API для поиска версий под 1.16.5 и fabric
        $apiUrl = "https://api.modrinth.com/v2/project/$modSlug/version?game_versions=[`"1.16.5`"]&loaders=[`"fabric`"]"
        $response = Invoke-RestMethod -Uri $apiUrl -UserAgent "Mozilla/5.0"
        
        if ($response -and $response.Count -gt 0) {
            # Берем первый подходящий файл из свежего релиза
            $fileInfo = $response[0].files | Where-Object { $_.primary -eq $true } | Select-Object -First 1
            if (-not $fileInfo) { $fileInfo = $response[0].files[0] }

            $downloadUrl = $fileInfo.url
            $fileName = $fileInfo.filename
            $outPath = Join-Path $targetFolder $fileName

            Invoke-WebRequest -Uri $downloadUrl -OutFile $outPath -UserAgent "Mozilla/5.0"
            Write-Host " [ГОТОВО] -> $fileName" -ForegroundColor Green
        } else {
            Write-Host " [НЕ НАЙДЕН ПОД 1.16.5 FABRIC]" -ForegroundColor Yellow
        }
    }
    catch {
        Write-Host " [ОШИБКА: $($_.Exception.Message)]" -ForegroundColor Red
    }
}

Write-Host "`n🎉 Скачивание завершено! Все файлы в папке Court_Modpack_1.16.5 на Рабочем столе." -ForegroundColor Green
