# Путь к папке на Рабочеме столе
$DesktopPath = [System.Environment]::GetFolderPath('Desktop')
$ModsFolder = Join-Path -Path $DesktopPath -ChildPath "Mods_Paid_Anomalies"

if (-not (Test-Path -Path $ModsFolder)) {
    New-Item -ItemType Directory -Path $ModsFolder | Out-Null
}

# Список недостающих библиотек (Fabric 1.20.1)
$MissingLibs = @(
    "resourcefullib",
    "malilib",
    "moonlight"
)

Write-Host "`nДокачиваем отсутствующие библиотеки (Fabric 1.20.1)...`n" -ForegroundColor Cyan

foreach ($slug in $MissingLibs) {
    try {
        $apiUrl = "https://api.modrinth.com/v2/project/$slug/version"
        $response = Invoke-RestMethod -Uri $apiUrl -Method Get -UserAgent "MinecraftModDownloader/1.0"
        
        # Ищем версию под Fabric и 1.20.1
        $matchedVersion = $response | Where-Object { 
            ($_.loaders -contains "fabric") -and ($_.game_versions -contains "1.20.1") 
        } | Select-Object -First 1
        
        if ($matchedVersion) {
            $file = $matchedVersion.files | Where-Object { $_.primary -eq $true } | Select-Object -First 1
            if (-not $file) { $file = $matchedVersion.files[0] }
            
            $downloadUrl = $file.url
            $fileName = $file.filename
            $outputPath = Join-Path -Path $ModsFolder -ChildPath $fileName

            Write-Host "Скачиваем: $fileName..." -NoNewline
            Invoke-WebRequest -Uri $downloadUrl -OutFile $outputPath
            Write-Host " [ГОТОВО]" -ForegroundColor Green
        } else {
            Write-Host "Не найден файл для $slug" -ForegroundColor Red
        }
    } catch {
        Write-Host "Ошибка при скачивании $slug : $_" -ForegroundColor Red
    }
}

Write-Host "`n Все библиотеки успешно скачаны!" -ForegroundColor Green
